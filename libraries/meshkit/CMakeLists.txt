cmake_minimum_required(VERSION 3.21)
project(MeshKit VERSION 0.2.0 LANGUAGES C CXX)

# The standard C++ event API is independent of the optional transitional Qt targets.
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(meshkit_runtime INTERFACE)
add_library(meshkit::runtime ALIAS meshkit_runtime)
set_target_properties(meshkit_runtime PROPERTIES EXPORT_NAME runtime)
target_compile_features(meshkit_runtime INTERFACE cxx_std_17)
target_include_directories(meshkit_runtime INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
find_package(Threads REQUIRED)
target_link_libraries(meshkit_runtime INTERFACE Threads::Threads)

include(cmake/CoreTargets.cmake)
option(MESHKIT_BUILD_CLI "Build the standard C++ mesh command-line tool" ${PROJECT_IS_TOP_LEVEL})
if(MESHKIT_BUILD_CLI)
    add_executable(meshkit_cli tools/meshkit.cpp)
    set_target_properties(meshkit_cli PROPERTIES OUTPUT_NAME meshkit AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(meshkit_cli PRIVATE meshkit::io)
    install(TARGETS meshkit_cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()

option(MESHKIT_BUILD_PYTHON "Build optional Python bindings without Qt or NumPy headers" OFF)
if(MESHKIT_BUILD_PYTHON)
    include(cmake/PythonTargets.cmake)
endif()

option(MESHKIT_BUILD_QT "Build the extracted mesher and codecs with their current Qt API" ON)
if(MESHKIT_BUILD_QT)
    include(cmake/QtTargets.cmake)
endif()

option(MESHKIT_BUILD_BENCHMARKS "Build opt-in standalone capacity probes" OFF)
if(MESHKIT_BUILD_BENCHMARKS)
    add_executable(meshkit_native_scale_probe tests/native_scale_probe.cpp)
    set_target_properties(meshkit_native_scale_probe PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(meshkit_native_scale_probe PRIVATE meshkit::io)
endif()
if(MESHKIT_BUILD_BENCHMARKS AND MESHKIT_BUILD_QT)
    add_executable(meshkit_scale_probe tests/scale_probe.cpp)
    set_target_properties(meshkit_scale_probe PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(meshkit_scale_probe PRIVATE meshkit::core_qt)
endif()

install(TARGETS meshkit_runtime EXPORT MeshKitTargets)
install(DIRECTORY include/meshkit DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(MESHKIT_BUILD_QT)
    install(DIRECTORY include/mesh include/core DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
endif()
install(FILES LICENSE NOTICE RELICENSING.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/MeshKit)
install(EXPORT MeshKitTargets NAMESPACE meshkit:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeshKit)
configure_package_config_file(cmake/MeshKitConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/MeshKitConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeshKit)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MeshKitConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MeshKitConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/MeshKitConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeshKit)

option(MESHKIT_BUILD_TESTS "Build standalone MeshKit tests" ${PROJECT_IS_TOP_LEVEL})

    add_executable(meshkit_runtime_test tests/runtime_test.cpp)

    add_executable(meshkit_native_mixed_test tests/native_mixed_test.cpp)

    target_link_libraries(meshkit_native_mixed_test PRIVATE meshkit::core)

    add_executable(meshkit_raster_test tests/raster_test.cpp)
    set_target_properties(meshkit_raster_test PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(meshkit_raster_test PRIVATE meshkit::core)
    add_test(NAME meshkit_raster_test COMMAND meshkit_raster_test)
    if(MESHKIT_BUILD_PYTHON)
        add_test(NAME meshkit_python_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/python_test.py)
        set_tests_properties(meshkit_python_test PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/python")
    endif()
    add_executable(meshkit_native_io_test tests/native_io_test.cpp)
    set_target_properties(meshkit_native_io_test PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(meshkit_native_io_test PRIVATE meshkit::io)
    add_test(NAME meshkit_native_io_test COMMAND meshkit_native_io_test)
    if(MESHKIT_BUILD_QT)
        find_package(Qt6 6.5 REQUIRED COMPONENTS Test)

        set_target_properties(meshkit_kernel_parity_test PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)

        set_target_properties(meshkit_constraints_test PROPERTIES AUTOMOC ON AUTOUIC OFF AUTORCC OFF)
        target_link_libraries(meshkit_constraints_test PRIVATE meshkit::core_qt Qt6::Test)
        add_test(NAME meshkit_constraints_test COMMAND meshkit_constraints_test)
        add_executable(meshkit_sizing_test tests/sizing_test.cpp)
        set_target_properties(meshkit_sizing_test PROPERTIES AUTOMOC ON AUTOUIC OFF AUTORCC OFF)
        target_link_libraries(meshkit_sizing_test PRIVATE meshkit::core_qt Qt6::Test)
        add_test(NAME meshkit_sizing_test COMMAND meshkit_sizing_test)
        add_executable(meshkit_cellplanner_test tests/cellplanner_test.cpp)
        set_target_properties(meshkit_cellplanner_test PROPERTIES AUTOMOC ON AUTOUIC OFF AUTORCC OFF)
        target_link_libraries(meshkit_cellplanner_test PRIVATE meshkit::core_qt Qt6::Test)
        add_test(NAME meshkit_cellplanner_test COMMAND meshkit_cellplanner_test)
        add_executable(meshkit_thinning_test tests/thinning_test.cpp)
        set_target_properties(meshkit_thinning_test PROPERTIES AUTOMOC ON AUTOUIC OFF AUTORCC OFF)
