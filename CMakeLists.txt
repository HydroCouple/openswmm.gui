cmake_minimum_required(VERSION 3.5)

project(
    SWMMVis
    VERSION 6.0.0
    LANGUAGES C CXX
    HOMEPAGE_URL "https://github.com/DigitalWaterAnalytics/SWMMVis"
    DESCRIPTION "A SWMM model setup, analysis, and visualization tool" 
)

configure_file(
    ${PROJECT_SOURCE_DIR}/include/version.h.in 
    ${PROJECT_SOURCE_DIR}/include/version.h
)

set(CMAKE_AUTOUIC ON) 
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

set(CMAKE_AUTOUIC_SEARCH_PATHS forms)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets OpenGL OpenGLWidgets LinguistTools)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets OpenGL OpenGLWidgets LinguistTools)

set(TS_FILES translation/SWMMVis_en_US.ts)


# The MACOSX_BUNDLE_ICON_FILE variable is added to the Info.plist
# generated by CMake. This variable contains the .icns file name,
# without the path.
set(MACOSX_BUNDLE_ICON_FILE swmmvis.icns)
# And the following tells CMake where to find and install the file itself.
set(app_icon_macos "${CMAKE_CURRENT_SOURCE_DIR}/resources/swmmvis.icns")
set_source_files_properties(${app_icon_macos} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")

## And the same for Windows
set(app_icon_resource_windows "${CMAKE_CURRENT_SOURCE_DIR}/resources/swmmvis.rc")


set(
    PROJECT_SOURCES

        # Header files
        include/swmmvis.h
        include/swmmvislayer.h
        include/swmmvisproject.h
        include/swmmvisapplication.h
        include/swmmvissplashscreen.h
        include/swmmvisscene.h
        include/swmmvisgraphicsview.h
        include/swmmvisprojectmodel.h
        include/swmmvisprojectwindow.h
        include/swmmvisservice.h
        include/swmmvislogmessage.h
        include/swmmvissubproject.h

        # Source files
        src/main.cpp
        src/swmmvis.cpp
        src/swmmvislayer.cpp
        src/swmmvisproject.cpp
        src/swmmvisapplication.cpp
        src/swmmvissplashscreen.cpp
        src/swmmvisscene.cpp
        src/swmmvisgraphicsview.cpp
        src/swmmvisprojectmodel.cpp
        src/swmmvisprojectwindow.cpp
        src/swmmvislogmessage.cpp
        src/swmmvissubproject.cpp

        # Form files
        forms/swmmvis.ui
        forms/mainwindow.ui

        # Translation files
        ${TS_FILES}

        # Resource files
        ${app_icon_macos}
        ${app_icon_resource_windows}
)

if(${QT_VERSION_MAJOR} GREATER_EQUAL 6)
    qt_add_executable(SWMMVis
        MANUAL_FINALIZATION
        ${PROJECT_SOURCES}
        resources/swmmvis.qrc
    )
# Define target properties for Android with Qt 6 as:
#    set_property(TARGET SWMMVis APPEND PROPERTY QT_ANDROID_PACKAGE_SOURCE_DIR
#                 ${CMAKE_CURRENT_SOURCE_DIR}/android)
# For more information, see https://doc.qt.io/qt-6/qt-add-executable.html#target-creation

qt_create_translation(QM_FILES ${CMAKE_SOURCE_DIR} ${TS_FILES})

else()
    if(ANDROID)
        add_library(SWMMVis SHARED
            ${PROJECT_SOURCES}
        )
# Define properties for Android with Qt 5 after find_package() calls as:
#    set(ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android")
    else()
        add_executable(SWMMVis
            ${PROJECT_SOURCES}
        )
    endif()

    qt5_create_translation(QM_FILES ${CMAKE_SOURCE_DIR} ${TS_FILES})
endif()

target_include_directories(
    SWMMVis 
        PRIVATE include
        PRIVATE forms
)

target_link_libraries(
    SWMMVis
        PRIVATE
        Qt${QT_VERSION_MAJOR}::Widgets
        Qt${QT_VERSION_MAJOR}::OpenGL
        Qt${QT_VERSION_MAJOR}::OpenGLWidgets
 )

set_target_properties(SWMMVis PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER swmmvis.epa.gov
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

install(TARGETS SWMMVis
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

if(QT_VERSION_MAJOR EQUAL 6)
    qt_finalize_executable(SWMMVis)
endif()
